# Urban 客户端验证机制设计

## 1. 概述

本文档定义 Urban 客户端的授权验证机制，包括离线验证和在线验证两种模式。

## 2. 客户端授权数据模型

### 2.1 SQLite 授权表设计

```sql
CREATE TABLE ClientAuthorization (
    Id INTEGER PRIMARY KEY AUTOINCREMENT,
    ProductCode TEXT NOT NULL,                    -- 产品代码
    AuthToken TEXT NOT NULL UNIQUE,               -- 授权令牌
    MachineCode TEXT NOT NULL,                     -- 绑定的机器码
    AuthType INTEGER NOT NULL,                     -- 0=离线, 1=在线
    AuthStatus INTEGER NOT NULL DEFAULT 1,         -- 0=失效, 1=正常
    AuthBeginDate TEXT NOT NULL,                   -- 授权开始时间
    AuthEndDate TEXT NOT NULL,                     -- 授权结束时间
    ProId TEXT,                                    -- 项目ID
    ProName TEXT,                                  -- 项目名称
    Signature TEXT,                                 -- 数字签名（离线授权）
    LastVerifyDate TEXT,                           -- 最后验证时间
    CreateDate TEXT NOT NULL,                      -- 创建时间
    UpdateDate TEXT                                -- 更新时间
);

CREATE INDEX idx_client_auth_token ON ClientAuthorization(AuthToken);
CREATE INDEX idx_client_auth_status ON ClientAuthorization(AuthStatus);
```

## 3. 机器码获取

### 3.1 机器码生成算法

```csharp
public class MachineCodeProvider
{
    private static readonly string[] Salt = {
        "FdSoft", "Urban", "Material", "2026"
    };

    public static string GetMachineCode()
    {
        var components = new List<string>();

        // 1. CPU ID
        components.Add(GetCpuId());

        // 2. 主板序列号
        components.Add(GetMotherboardSerial());

        // 3. MAC 地址（取第一个网卡）
        components.Add(GetMacAddress());

        // 4. 硬盘序列号（系统盘）
        components.Add(GetDiskSerial());

        // 组合并哈希
        var combined = string.Join("|", components);
        using var sha256 = SHA256.Create();
        var hash = sha256.ComputeHash(Encoding.UTF8.GetBytes(combined));

        // 转换为十六进制字符串，取前 32 位
        return BitConverter.ToString(hash).Replace("-", "").Substring(0, 32);
    }

    private static string GetCpuId()
    {
        try
        {
            var cpu = new ManagementObjectSearcher("select ProcessorId from Win32_Processor")
                .Get().Cast<ManagementObject>().FirstOrDefault();
            return cpu?["ProcessorId"]?.ToString() ?? "UNKNOWN_CPU";
        }
        catch
        {
            return "FALLBACK_CPU";
        }
    }

    private static string GetMotherboardSerial()
    {
        try
        {
            var board = new ManagementObjectSearcher("select SerialNumber from Win32_BaseBoard")
                .Get().Cast<ManagementObject>().FirstOrDefault();
            return board?["SerialNumber"]?.ToString() ?? "UNKNOWN_BOARD";
        }
        catch
        {
            return "FALLBACK_BOARD";
        }
    }

    private static string GetMacAddress()
    {
        try
        {
            var nic = NetworkInterface.GetAllNetworkInterfaces()
                .FirstOrDefault(n => n.NetworkInterfaceType == NetworkInterfaceType.Ethernet
                    && n.OperationalStatus == OperationalStatus.Up
                    && !n.Description.Contains("Virtual"));
            return nic?.GetPhysicalAddress().ToString() ?? "UNKNOWN_MAC";
        }
        catch
        {
            return "FALLBACK_MAC";
        }
    }

    private static string GetDiskSerial()
    {
        try
        {
            var disk = new ManagementObjectSearcher("select SerialNumber from Win32_DiskDrive WHERE Index=0")
                .Get().Cast<ManagementObject>().FirstOrDefault();
            return disk?["SerialNumber"]?.ToString() ?? "UNKNOWN_DISK";
        }
        catch
        {
            return "FALLBACK_DISK";
        }
    }
}
```

## 4. 离线验证机制

### 4.1 启动时验证流程

```csharp
public class OfflineAuthValidator
{
    private readonly IDbContextFactory<AppDbContext> _dbFactory;
    private readonly ILogger<OfflineAuthValidator> _logger;

    public async Task<ValidationResult> ValidateOnStartup()
    {
        // 1. 获取当前机器码
        var currentMachineCode = MachineCodeProvider.GetMachineCode();
        _logger.LogDebug("Current machine code: {MachineCode}", currentMachineCode);

        // 2. 从数据库读取授权信息
        await using var db = await _dbFactory.CreateDbContextAsync();
        var authorization = await db.ClientAuthorizations
            .Where(a => a.ProductCode == "UrbanManagement")
            .Where(a => a.AuthStatus == AuthStatus.Valid)
            .OrderByDescending(a => a.CreateDate)
            .FirstOrDefaultAsync();

        if (authorization == null)
        {
            _logger.LogWarning("No valid authorization found");
            return ValidationResult.Fail("未找到有效授权，请先导入授权文件或使用授权码激活");
        }

        // 3. 检查授权是否过期
        if (DateTime.UtcNow > authorization.AuthEndDate)
        {
            _logger.LogWarning("Authorization expired at {EndDate}", authorization.AuthEndDate);
            authorization.AuthStatus = AuthStatus.Invalid;
            await db.SaveChangesAsync();
            return ValidationResult.Fail("授权已过期，请联系管理员续期");
        }

        // 4. 验证机器码是否匹配（关键步骤！）
        if (currentMachineCode != authorization.MachineCode)
        {
            _logger.LogError("Machine code mismatch: Expected={Expected}, Actual={Actual}",
                authorization.MachineCode, currentMachineCode);
            authorization.AuthStatus = AuthStatus.Invalid;
            await db.SaveChangesAsync();
            return ValidationResult.Fail("授权无效：机器码不匹配，请联系管理员");
        }

        // 5. 验证签名（仅离线授权）
        if (authorization.AuthType == AuthType.Offline)
        {
            if (!VerifySignature(authorization))
            {
                _logger.LogError("License signature verification failed");
                authorization.AuthStatus = AuthStatus.Invalid;
                await db.SaveChangesAsync();
                return ValidationResult.Fail("授权文件已被篡改，请重新导入");
            }
        }

        // 6. 更新最后验证时间
        authorization.LastVerifyDate = DateTime.UtcNow;
        await db.SaveChangesAsync();

        _logger.LogInformation("Authorization validated successfully: {Token}", authorization.AuthToken);
        return ValidationResult.Success(authorization);
    }

    private bool VerifySignature(ClientAuthorization auth)
    {
        // 使用公钥验证签名
        var content = GetSignatureContent(auth);
        var rsa = GetPublicKey();
        return LicenseSignature.Verify(content, auth.Signature!, rsa);
    }
}
```

### 4.2 授权文件导入

```csharp
public class LicenseFileImporter
{
    private readonly IDbContextFactory<AppDbContext> _dbFactory;
    private readonly ILogger<LicenseFileImporter> _logger;

    public async Task<ImportResult> ImportAsync(string filePath, string password)
    {
        try
        {
            // 1. 读取并解密文件
            var encryptedContent = await File.ReadAllTextAsync(filePath);
            var json = LicenseFileEncryption.Decrypt(encryptedContent, password);

            // 2. 解析授权数据
            var licenseData = JsonSerializer.Deserialize<LicenseData>(json);

            // 3. 验证签名
            if (!VerifyLicenseSignature(licenseData))
            {
                return ImportResult.Fail("授权文件签名验证失败");
            }

            // 4. 验证机器码
            var currentMachineCode = MachineCodeProvider.GetMachineCode();
            if (licenseData.MachineCode != currentMachineCode)
            {
                return ImportResult.Fail($"授权文件机器码不匹配。授权文件：{licenseData.MachineCode}，当前机器：{currentMachineCode}");
            }

            // 5. 写入数据库
            await using var db = await _dbFactory.CreateDbContextAsync();

            // 检查是否已存在相同授权
            var existing = await db.ClientAuthorizations
                .FirstOrDefaultAsync(a => a.AuthToken == licenseData.AuthToken);

            if (existing != null)
            {
                // 更新现有授权
                existing.AuthStatus = AuthStatus.Valid;
                existing.UpdateDate = DateTime.UtcNow;
            }
            else
            {
                // 插入新授权
                var auth = new ClientAuthorization
                {
                    ProductCode = licenseData.ProductCode,
                    AuthToken = licenseData.AuthToken,
                    MachineCode = licenseData.MachineCode,
                    AuthType = AuthType.Offline,
                    AuthStatus = AuthStatus.Valid,
                    AuthBeginDate = licenseData.AuthBeginDate,
                    AuthEndDate = licenseData.AuthEndDate,
                    ProId = licenseData.ProId,
                    ProName = licenseData.ProName,
                    Signature = licenseData.Signature,
                    CreateDate = DateTime.UtcNow
                };
                await db.ClientAuthorizations.AddAsync(auth);
            }

            await db.SaveChangesAsync();
            _logger.LogInformation("License imported successfully: {Token}", licenseData.AuthToken);

            return ImportResult.Success("授权文件导入成功");
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "Failed to import license file");
            return ImportResult.Fail($"导入失败：{ex.Message}");
        }
    }
}
```

## 5. 在线验证机制

### 5.1 授权码激活

```csharp
public class OnlineAuthActivator
{
    private readonly IBasePlatformAuthApi _authApi;
    private readonly IDbContextFactory<AppDbContext> _dbFactory;
    private readonly ILogger<OnlineAuthActivator> _logger;

    public async Task<ActivationResult> ActivateAsync(string authCode)
    {
        try
        {
            // 1. 获取当前机器码
            var currentMachineCode = MachineCodeProvider.GetMachineCode();

            // 2. 调用 BasePlatform API 激活
            var request = new ActivateRequest
            {
                ProductCode = "UrbanManagement",
                Code = authCode,
                MachineCode = currentMachineCode
            };

            var response = await _authApi.ActivateAsync(request);

            if (!response.Success)
            {
                return ActivationResult.Fail(response.Message ?? "激活失败");
            }

            // 3. 将授权信息写入 SQLite
            await using var db = await _dbFactory.CreateDbContextAsync();

            var auth = new ClientAuthorization
            {
                ProductCode = "UrbanManagement",
                AuthToken = response.Data.AuthToken,
                MachineCode = currentMachineCode,
                AuthType = AuthType.Online,
                AuthStatus = AuthStatus.Valid,
                AuthBeginDate = DateTime.UtcNow,
                AuthEndDate = response.Data.AuthEndDate,
                ProId = response.Data.ProId,
                ProName = response.Data.ProName,
                LastVerifyDate = DateTime.UtcNow,
                CreateDate = DateTime.UtcNow
            };

            await db.ClientAuthorizations.AddAsync(auth);
            await db.SaveChangesAsync();

            _logger.LogInformation("Authorization activated successfully: {Token}", response.Data.AuthToken);
            return ActivationResult.Success("激活成功");
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "Failed to activate authorization");
            return ActivationResult.Fail($"激活失败：{ex.Message}");
        }
    }
}
```

### 5.2 定期在线验证

```csharp
public class OnlineAuthVerifier : BackgroundService
{
    private readonly IBasePlatformAuthApi _authApi;
    private readonly IDbContextFactory<AppDbContext> _dbFactory;
    private readonly ILogger<OnlineAuthVerifier> _logger;

    protected override async Task ExecuteAsync(CancellationToken stoppingToken)
    {
        // 每天验证一次
        while (!stoppingToken.IsCancellationRequested)
        {
            try
            {
                await VerifyAsync();

                // 计算下次验证时间（24 小时后）
                await Task.Delay(TimeSpan.FromHours(24), stoppingToken);
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "Online verification failed");
                // 出错后 1 小时重试
                await Task.Delay(TimeSpan.FromHours(1), stoppingToken);
            }
        }
    }

    public async Task<VerificationResult> VerifyAsync()
    {
        // 1. 获取当前机器码
        var currentMachineCode = MachineCodeProvider.GetMachineCode();

        // 2. 从数据库读取授权信息
        await using var db = await _dbFactory.CreateDbContextAsync();
        var authorization = await db.ClientAuthorizations
            .Where(a => a.ProductCode == "UrbanManagement")
            .Where(a => a.AuthStatus == AuthStatus.Valid)
            .OrderByDescending(a => a.CreateDate)
            .FirstOrDefaultAsync();

        if (authorization == null)
        {
            return VerificationResult.Fail("无有效授权");
        }

        // 3. 调用 BasePlatform API 验证
        var request = new VerifyRequest
        {
            ProductCode = "UrbanManagement",
            AuthToken = authorization.AuthToken,
            MachineCode = currentMachineCode
        };

        var response = await _authApi.VerifyAsync(request);

        if (!response.Success)
        {
            // API 验证失败，标记授权为失效
            authorization.AuthStatus = AuthStatus.Invalid;
            await db.SaveChangesAsync();
            return VerificationResult.Fail("服务器验证失败");
        }

        // 4. 三方比对：当前机器码 == SQLite 机器码 == 服务器机器码
        if (!response.Data.IsValid)
        {
            authorization.AuthStatus = AuthStatus.Invalid;
            await db.SaveChangesAsync();
            return VerificationResult.Fail("服务器验证失败：授权无效");
        }

        if (!response.Data.MachineCodeMatch)
        {
            authorization.AuthStatus = AuthStatus.Invalid;
            await db.SaveChangesAsync();
            return VerificationResult.Fail("服务器验证失败：机器码不匹配");
        }

        if (currentMachineCode != authorization.MachineCode)
        {
            authorization.AuthStatus = AuthStatus.Invalid;
            await db.SaveChangesAsync();
            return VerificationResult.Fail("本地验证失败：机器码不匹配");
        }

        // 5. 更新最后验证时间
        authorization.LastVerifyDate = DateTime.UtcNow;
        await db.SaveChangesAsync();

        return VerificationResult.Success();
    }
}
```

### 5.3 手动触发验证

```csharp
public class ManualAuthVerifier
{
    private readonly OnlineAuthVerifier _verifier;

    public async Task<bool> VerifyNow()
    {
        var result = await _verifier.VerifyAsync();
        if (!result.IsValid)
        {
            MessageBox.Show(result.Message, "验证失败", MessageBoxButton.OK, MessageBoxImage.Error);
            // 如果授权失效，关闭程序
            if (result.IsAuthorizationInvalid)
            {
                Application.Current.Shutdown();
            }
            return false;
        }
        MessageBox.Show("验证成功", "验证", MessageBoxButton.OK, MessageBoxImage.Information);
        return true;
    }
}
```

### 6.1 授权失效策略

```csharp
public class AuthFailureHandler
{
    public void HandleAuthFailure(ValidationResult result)
    {
        var message = $"授权无效：{result.Message}\n\n请联系管理员获取授权。";
        var result = MessageBox.Show(
            message,
            "授权验证失败",
            MessageBoxButton.OK,
            MessageBoxImage.Stop);

        // 无论用户选择什么，都关闭程序
        Application.Current.Shutdown();
    }
```

### 6.2 宽限期机制（可选）

```csharp
// 对于网络中断情况，提供离线宽限期
public class AuthGracePeriod
{
    private const int GraceDays = 7; // 7 天宽限期

    public bool IsInGracePeriod(ClientAuthorization auth)
    {
        if (auth.LastVerifyDate == null) return false;

        var daysSinceLastVerify = (DateTime.UtcNow - auth.LastVerifyDate.Value).TotalDays;
        return daysSinceLastVerify <= GraceDays;
    }
}
```
